   158→// ── Helpers ────────────────────────────────────────────────────────────────────
   159→
   160→const inrFormat = (amount: number) =>
   161→  `₹${new Intl.NumberFormat('en-IN').format(amount)}`;
   162→
   163→const formatDate = (dateStr: string) => {
   164→  if (!dateStr) return '—';
   165→  return new Date(dateStr).toLocaleDateString('en-IN', {
   166→    day: '2-digit',
   167→    month: 'short',
   168→    year: 'numeric',
   169→  });
   170→};
   171→
   172→const formatDateTime = (dateStr: string, timeStr?: string) => {
   173→  if (!dateStr) return '—';
   174→  const date = new Date(dateStr);
   175→  const formattedDate = date.toLocaleDateString('en-IN', {
   176→    day: '2-digit',
   177→    month: 'short',
   178→    year: 'numeric',
   179→  });
   180→  if (timeStr) {
   181→    return `${formattedDate}, ${timeStr}`;
   182→  }
   183→  return formattedDate;
   184→};
   185→
   186→const getStatusBadge = (status: string) => {
   187→  const s = status?.toLowerCase();
   188→  const map: Record<string, string> = {
   189→    scheduled: 'bg-blue-100 text-blue-800 hover:bg-blue-100',
   190→    confirmed: 'bg-green-100 text-green-800 hover:bg-green-100',
   191→    completed: 'bg-teal-100 text-teal-800 hover:bg-teal-100',
   192→    cancelled: 'bg-red-100 text-red-800 hover:bg-red-100',
   193→    'no-show': 'bg-orange-100 text-orange-800 hover:bg-orange-100',
   194→    paid: 'bg-green-100 text-green-800 hover:bg-green-100',
   195→    partial: 'bg-amber-100 text-amber-800 hover:bg-amber-100',
   196→    pending: 'bg-red-100 text-red-800 hover:bg-red-100',
   197→  };
   198→  return map[s] || 'bg-gray-100 text-gray-800 hover:bg-gray-100';
   199→};
