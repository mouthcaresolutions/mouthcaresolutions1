   102→function formatCurrency(amount: number): string {
   103→  return new Intl.NumberFormat("en-IN", {
   104→    style: "currency",
   105→    currency: "INR",
   106→    minimumFractionDigits: 0,
   107→    maximumFractionDigits: 0,
   108→  }).format(amount);
   109→}
   110→
   111→function formatDate(dateStr: string | null): string {
   112→  if (!dateStr) return "\u2014";
   113→  try {
   114→    return new Date(dateStr).toLocaleDateString("en-IN", {
   115→      day: "2-digit",
   116→      month: "short",
   117→      year: "numeric",
   118→    });
   119→  } catch {
   120→    return dateStr;
   121→  }
   122→}
   123→
   124→function getCategoryBadge(category: string | null) {
   125→  if (!category) return <span className="text-sm text-gray-400">\u2014</span>;
   126→  const catMap: Record<string, string> = {
   127→    new: "bg-emerald-100 text-emerald-700 border-emerald-200",
   128→    regular: "bg-blue-100 text-blue-700 border-blue-200",
   129→    vip: "bg-amber-100 text-amber-700 border-amber-200",
   130→  };
   131→  const cls = catMap[category.toLowerCase()] ?? "bg-gray-100 text-gray-600 border-gray-200";
   132→  return (
   133→    <Badge variant="outline" className={cls}>
   134→      {category}
   135→    </Badge>
   136→  );
   137→}
   138→
   139→function calcAgeFromDOB(dob: string): number | null {
   140→  if (!dob) return null;
   141→  const birth = new Date(dob);
   142→  const today = new Date();
   143→  let age = today.getFullYear() - birth.getFullYear();
   144→  const m = today.getMonth() - birth.getMonth();
   145→  if (m < 0 || (m === 0 && today.getDate() < birth.getDate())) age--;
   146→  return age > 0 ? age : null;
   147→}
