   108→
   109→function getToday(): string {
   110→  return new Date().toISOString().split("T")[0];
   111→}
   112→
   113→function formatDate(dateStr: string): string {
   114→  if (!dateStr) return "";
   115→  const d = new Date(dateStr + "T00:00:00");
   116→  return d.toLocaleDateString("en-IN", {
   117→    weekday: "short",
   118→    day: "numeric",
   119→    month: "short",
   120→    year: "numeric",
   121→  });
   122→}
   123→
   124→function formatTime(timeStr: string): string {
   125→  if (!timeStr) return "";
   126→  const [h, m] = timeStr.split(":");
   127→  const hour = parseInt(h, 10);
   128→  const ampm = hour >= 12 ? "PM" : "AM";
   129→  const h12 = hour % 12 || 12;
   130→  return `${h12}:${m} ${ampm}`;
   131→}
   132→
   133→const STATUS_CYCLE: Record<string, string[]> = {
   134→  scheduled: ["confirmed", "cancelled"],
   135→  confirmed: ["completed", "no-show", "cancelled"],
   136→  completed: ["scheduled"],
   137→  cancelled: ["scheduled"],
   138→  "no-show": ["scheduled"],
   139→};
   140→
   141→function getStatusBadge(status: string) {
   142→  const map: Record<string, string> = {
   143→    scheduled: "bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-200",
   144→    confirmed: "bg-blue-100 text-blue-700 border-blue-200 hover:bg-blue-200",
   145→    completed: "bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-200",
   146→    cancelled: "bg-red-100 text-red-700 border-red-200 hover:bg-red-200",
   147→    "no-show": "bg-gray-100 text-gray-600 border-gray-300 hover:bg-gray-200",
   148→  };
   149→  const cls = map[status] ?? "bg-gray-100 text-gray-600 border-gray-300";
   150→  return (
   151→    <Badge variant="outline" className={`${cls} cursor-pointer select-none transition-colors`}>
   152→      {status.charAt(0).toUpperCase() + status.slice(1)}
   153→    </Badge>
   154→  );
   155→}
   156→
   157→function getStatusNext(status: string): string {
   158→  return STATUS_CYCLE[status]?.[0] ?? "scheduled";
   159→}
   160→
   161→// Group appointments by date
   162→function groupByDate(appointments: Appointment[]): Record<string, Appointment[]> {
   163→  const groups: Record<string, Appointment[]> = {};
   164→  for (const apt of appointments) {
   165→    if (!groups[apt.date]) groups[apt.date] = [];
   166→    groups[apt.date].push(apt);
   167→  }
   168→  return groups;
   169→}
