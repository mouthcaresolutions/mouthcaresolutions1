   924→function PlatformConfigCard({ platform, config, icon, color, fields, onSave, onTest, loading, setupGuide, isWhatsApp = false }: {
   925→  platform: string;
   926→  config: any;
   927→  icon: string;
   928→  color: string;
   929→  fields: { key: string; label: string; placeholder: string; type?: string }[];
   930→  onSave: (platform: string, updates: any) => void;
   931→  onTest: (platform: string) => void;
   932→  loading: boolean;
   933→  setupGuide?: string;
   934→  isWhatsApp?: boolean;
   935→}) {
   936→  const [showGuide, setShowGuide] = useState(false);
   937→  const [formValues, setFormValues] = useState<Record<string, string>>({});
   938→
   939→  const displayName = platform === 'google_business' ? 'Google Business Profile' : platform.charAt(0).toUpperCase() + platform.slice(1);
   940→  const enabled = config?.enabled || false;
   941→
   942→  return (
   943→    <div className="border border-gray-200 rounded-xl overflow-hidden">
   944→      <div className={`flex items-center justify-between p-4 ${enabled ? color + '/10' : 'bg-gray-50'}`}>
   945→        <div className="flex items-center gap-3">
   946→          <span className="text-2xl">{icon}</span>
   947→          <div>
   948→            <h4 className="font-semibold text-gray-900">{displayName}</h4>
   949→            <div className="flex items-center gap-3 mt-0.5">
   950→              <span className={`text-xs px-2 py-0.5 rounded-full ${enabled ? 'bg-green-100 text-green-700' : 'bg-gray-200 text-gray-500'}`}>
   951→                {enabled ? 'Connected' : 'Not Connected'}
   952→              </span>
   953→              {config?.totalPosts > 0 && <span className="text-xs text-gray-400">{config.totalPosts} posts shared</span>}
   954→              {config?.lastPostedAt && <span className="text-xs text-gray-400">Last: {new Date(config.lastPostedAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' })}</span>}
   955→            </div>
   956→          </div>
   957→        </div>
   958→        <div className="flex items-center gap-3">
   959→          <Switch checked={enabled} onCheckedChange={(v) => onSave(platform, { enabled: v })} />
   960→        </div>
   961→      </div>
   962→
   963→      {enabled && !isWhatsApp && (
   964→        <div className="p-4 border-t border-gray-100 bg-white space-y-3">
   965→          {fields.map(field => (
   966→            <div key={field.key}>
   967→              <Label className="text-xs text-gray-500 flex items-center gap-1"><Key className="h-3 w-3" /> {field.label}</Label>
   968→              <Input
   969→                type={field.type || 'text'}
   970→                className="mt-1 font-mono text-sm"
   971→                placeholder={field.placeholder}
   972→                defaultValue={field.type === 'password' ? '' : (config?.[field.key] || '')}
   973→                onChange={(e) => setFormValues(prev => ({ ...prev, [field.key]: e.target.value }))}
   974→              />
   975→            </div>
   976→          ))}
   977→          {config?.lastError && (
   978→            <div className="flex items-start gap-2 p-2 bg-red-50 rounded-lg text-xs text-red-600">
   979→              <AlertCircle className="h-3.5 w-3.5 mt-0.5 shrink-0" />
   980→              <span>{config.lastError}</span>
   981→            </div>
   982→          )}
   983→          <div className="flex gap-2 pt-2">
   984→            <Button size="sm" onClick={() => {
   985→              const updates: any = { enabled: true };
   986→              Object.entries(formValues).forEach(([k, v]) => { if (v) updates[k] = v; });
   987→              onSave(platform, updates);
   988→            }} disabled={loading} className="bg-teal-600 hover:bg-teal-700 text-white">
   989→              {loading ? <Loader2 className="h-3.5 w-3.5 mr-1 animate-spin" /> : <Save className="h-3.5 w-3.5 mr-1" />}
   990→              Save & Connect
   991→            </Button>
   992→            <Button size="sm" variant="outline" onClick={() => onTest(platform)} disabled={loading}>
   993→              <Send className="h-3.5 w-3.5 mr-1" /> Test Post
   994→            </Button>
   995→            {setupGuide && (
   996→              <Button size="sm" variant="ghost" onClick={() => setShowGuide(!showGuide)}>
   997→                <Eye className="h-3.5 w-3.5 mr-1" /> {showGuide ? 'Hide' : 'Setup'} Guide
   998→              </Button>
   999→            )}
  1000→          </div>
  1001→          {showGuide && setupGuide && (
  1002→            <pre className="mt-3 p-3 bg-gray-50 rounded-lg text-xs text-gray-600 whitespace-pre-wrap font-mono border">
  1003→              {setupGuide}
  1004→            </pre>
  1005→          )}
  1006→        </div>
  1007→      )}
  1008→
  1009→      {enabled && isWhatsApp && (
  1010→        <div className="p-4 border-t border-gray-100 bg-white">
  1011→          <p className="text-sm text-gray-600">WhatsApp does not support automated posting via API. Blog posts will generate <strong>shareable WhatsApp links</strong> that you can use in your WhatsApp Status, Groups, or broadcast lists.</p>
  1012→          <p className="text-xs text-gray-400 mt-2">Every blog post automatically gets a WhatsApp share link. You can also share manually from the Posts tab.</p>
  1013→        </div>
  1014→      )}
  1015→    </div>
  1016→  );
